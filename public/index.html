Kirim pesan dari web e-commerce

POST /api/send · header Authorization: TOKEN · body JSON target, message, url (opsional)

PHP 
$ch = curl_init('/api/send');
curl_setopt_array($ch, [
  CURLOPT_POST => true,
  CURLOPT_RETURNTRANSFER => true,
  CURLOPT_TIMEOUT => 5,
  CURLOPT_HTTPHEADER => ['Authorization: TOKEN_ANDA', 'Content-Type: application/json'],
  CURLOPT_POSTFIELDS => json_encode([
    'target'  => $order->no_hp,
    'message' => "Halo {$order->nama}, pesanan #{$order->id} sudah kami terima. Total: Rp {$order->total}",
  ]),
]);
$res = curl_exec($ch); // jangan sampai error WA menggagalkan checkout
curl_close($ch);
JavaScript 
await fetch('/api/send', {
  method: 'POST',
  headers: { Authorization: 'TOKEN_ANDA', 'Content-Type': 'application/json' },
  body: JSON.stringify({ target: '08123456789', message: 'Pesanan Anda sudah dikirim 🚚' })
});
